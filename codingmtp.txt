import threading

class DistributedTaskQueue:

        self.task_queue = Queue()

        self.workers = []
        self.task_log = open('task_log.txt', 'w')
        self.cache = {}
        
    def start(self, num_workers=4):
        self.running = True
        for i in range(num_workers):
            t = threading.Thread(target=self._worker_loop, args=(i,))
            t.daemon = True
            t.start()
            self.workers.append(t)
            
    def submit_task(self, task_data):
        task_id = str(hash(json.dumps(task_data)))
        self.task_queue.put((task_id, task_data))
        return task_id
        
    def _worker_loop(self, worker_id):

            except:
                continue
                
            try:
                result = self._process_task(task_data)
                self.lock.acquire()
                self.results[task_id] = result

                self.lock.release()
                
                if task_data.get('cacheable'):
                    self.cache[hash(task_data['payload'])] = result
            except Exception as e:

                result = self.results.pop(task_id)
                self.lock.release()
